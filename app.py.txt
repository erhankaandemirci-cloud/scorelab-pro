# 2. ARKA PLAN MOTORLARI (Dixon-Coles & Kelly)
# ==========================================================
class DixonColesMotoru:
    def matris_uret(self, le, ld, max_g=15, rho=-0.08):
        le, ld = float(np.clip(le, 0.15, 6.0)), float(np.clip(ld, 0.15, 6.0))
        alt_sinir = max(-1.0 / max(le * ld, 1e-6) + 1e-5, -0.99)
        ust_sinir = min(1.0 / max(le, 1e-6) - 1e-5, 1.0 / max(ld, 1e-6) - 1e-5, 0.99)
        rho = float(np.clip(rho, alt_sinir, ust_sinir))

        g = np.arange(max_g)
        mat = np.outer(poisson.pmf(g, le), poisson.pmf(g, ld))
        mat[0, 0] *= (1.0 - le * ld * rho)
        mat[0, 1] *= (1.0 + le * rho)
        mat[1, 0] *= (1.0 + ld * rho)
        mat[1, 1] *= (1.0 - rho)
        return np.clip(mat, 0.0, None) / np.sum(np.clip(mat, 0.0, None))

class FinansalRiskMotoru:
    def yatırım_miktari_hesapla(self, kasa, guvenli_faktor, p, oran):
        b = oran - 1.0
        if b <= 0: return 0.0, 0.0, "Geçersiz Oran"
        tam_kelly = (p * b - (1.0 - p)) / b
        if tam_kelly <= 0: return 0.0, 0.0, "🔴 Değer Yok (Pas)"
        fraksiyonel_oran = tam_kelly * guvenli_faktor
        tutar = kasa * fraksiyonel_oran
        return round(fraksiyonel_oran * 100, 2), round(tutar, 2), "🟢 OYNA"

analiz_motoru = DixonColesMotoru()
risk_motoru = FinansalRiskMotoru()

# 3. GÜVENLİ CANLI VERİ ENTEGRASYONU (API)
@st.cache_data(ttl=3600, show_spinner=False)
def canli_verileri_getir():
    try:
        # Şifreni ileride site ayarlarından gireceğiz
        api_key = st.secrets["API_KEY"]
    except:
        return pd.DataFrame(), "⚠️ HATA: API Anahtarını site ayarlarından eklemediniz."
        
    mac_listesi = []
    url = f"https://api.the-odds-api.com/v4/sports/soccer_epl/odds?apiKey={api_key}&regions=eu&markets=h2h&oddsFormat=decimal"
    
    try:
        response = requests.get(url, timeout=12)
        if response.status_code == 200:
            for mac in response.json():
                ev, dep = mac['home_team'], mac['away_team']
                o1, ox, o2 = 0.0, 0.0, 0.0
                for bm in mac.get('bookmakers', []):
                    if bm['key'] in ['bet365', 'pinnacle']:
                        oranlar = bm['markets'][0]['outcomes']
                        for o in oranlar:
                            if o['name'] == ev: o1 = o['price']
                            elif o['name'] == 'Draw': ox = o['price']
                            elif o['name'] == dep: o2 = o['price']
                        break 
                if o1 > 0 and ox > 0 and o2 > 0:
                    mac_listesi.append({'Ev': ev, 'Dep': dep, 'O1': o1, 'OX': ox, 'O2': o2})
    except:
        pass
        
    return pd.DataFrame(mac_listesi), "Senkronizasyon Tamam."

def xg_simulasyonu_uret():
    # Gerçek veri tabanı bağlanana kadar sistemi test etmek için
    return np.random.uniform(1.1, 2.7), np.random.uniform(0.7, 1.8)

    yuzde = int(olasilik * 100)
    return f"""

            <span style="font-family: sans-serif; font-weight: bold; color: #2c3e50;">{baslik}</span>
            <span style="font-weight: bold; color: {renk};">%{yuzde}</span>
        </div>
        <div style="width: 100%; background-color: #ecf0f1; border-radius: 6px; height: 18px;">
            <div style="width: {yuzde}%; background-color: {renk}; height: 18px; border-radius: 6px;"></div>
        </div>
    </div>
    """
